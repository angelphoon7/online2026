# RESHUFFLE Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a multi-party conditional exchange for event tickets on Arc Testnet, where participants sign the outcome they will accept and settlement executes only when every signed condition holds.

**Architecture:** Four Solidity contracts (TicketNFT, Escrow, IntentRegistry, Settlement) enforce conditions on-chain. A TypeScript solver reads the intent pool, searches for valid reshuffles, simulates, and submits. A Next.js frontend lets one operator play all three roles (swapper, buyer, seller) and reproduce every demo scene.

**Tech Stack:** Solidity 0.8.24 / Foundry, OpenZeppelin ERC-721, TypeScript / Node, Next.js 16 / React 19 / Tailwind, viem / wagmi, The Graph (stretch), Arc Testnet (EVM, USDC native gas).

## Global Constraints

- Solidity 0.8.x with checked arithmetic. No `unchecked` in validation paths.
- Named custom errors everywhere. Never `require` with a string.
- Validation order: V1-V8, unconditional, checks-effects-interactions.
- Bitmaps for session/section masks. No array loops where `&` works.
- `exactCount` not `minCount`. `mustShareSection` is separate from `sectionMask`.
- `maxNetPay` is a signed integer. Positive = debit ceiling, negative = credit floor.
- EIP-712 domain includes `chainId` and `verifyingContract`.
- Integer USDC, no rounding tolerance. Payment sums to exactly zero.
- Every rejection test asserts the specific named error, never bare `vm.expectRevert()`.
- No figure appears in any document unless it came from `forge test --gas-report`.
- Never write "optimal", "best price", "no grief risk", or "eliminates the failure mode".

```solidity
// src/TicketNFT.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts/token/ERC721/ERC721.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

contract TicketNFT is ERC721, Ownable {
    struct TicketMeta {
        uint32 eventId;
        uint16 sessionId;
        uint16 sectionId;
        uint16 row;
        uint16 seat;
        uint8  status; // 0 active, 1 redeemed
    }

    mapping(uint256 => TicketMeta) public meta;
    mapping(address => bool) public registeredIssuers;
    uint256 private _nextTokenId;

    error NotRegisteredIssuer(address caller);
    error NotTicketHolder(uint256 tokenId, address caller);
    error TicketAlreadyRedeemed(uint256 tokenId);

    event TicketMinted(uint256 indexed tokenId, address indexed to, uint32 eventId, uint16 sessionId, uint16 sectionId, uint16 row, uint16 seat);
    event TicketRedeemed(uint256 indexed tokenId, address indexed holder);
    event IssuerRegistered(address indexed issuer);
    event IssuerRemoved(address indexed issuer);

    constructor() ERC721("RESHUFFLE Ticket", "RTKT") Ownable(msg.sender) {}

    function registerIssuer(address issuer) external onlyOwner {
        registeredIssuers[issuer] = true;
        emit IssuerRegistered(issuer);
    }

    function removeIssuer(address issuer) external onlyOwner {
        registeredIssuers[issuer] = false;
        emit IssuerRemoved(issuer);
    }

    function mint(address to, TicketMeta calldata m) external returns (uint256) {
        if (!registeredIssuers[msg.sender]) revert NotRegisteredIssuer(msg.sender);

        uint256 tokenId = _nextTokenId++;
        _mint(to, tokenId);
        meta[tokenId] = m;

        emit TicketMinted(tokenId, to, m.eventId, m.sessionId, m.sectionId, m.row, m.seat);
        return tokenId;
    }

    function redeem(uint256 tokenId) external {
        if (ownerOf(tokenId) != msg.sender) revert NotTicketHolder(tokenId, msg.sender);
        if (meta[tokenId].status == 1) revert TicketAlreadyRedeemed(tokenId);

        meta[tokenId].status = 1;
        emit TicketRedeemed(tokenId, msg.sender);
    }
}
```

```solidity
// src/Escrow.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "./TicketNFT.sol";

contract Escrow {
    TicketNFT public immutable ticketNFT;
    address public settlement;

    mapping(uint256 => address) public depositor;

    error NotDepositor(uint256 tokenId, address caller);
    error TicketRedeemed(uint256 tokenId);
    error NotSettlement(address caller);
    error SettlementAlreadySet();

    event TicketEscrowed(uint256 indexed tokenId, address indexed depositor);
    event TicketWithdrawn(uint256 indexed tokenId, address indexed depositor);

    constructor(address _ticketNFT) {
        ticketNFT = TicketNFT(_ticketNFT);
    }

    function setSettlement(address _settlement) external {
        if (settlement != address(0)) revert SettlementAlreadySet();
        settlement = _settlement;
    }

    function deposit(uint256[] calldata ids) external {
        for (uint256 i = 0; i < ids.length; i++) {
            uint256 id = ids[i];
            (,,,,, uint8 status) = ticketNFT.meta(id);
            if (status == 1) revert TicketRedeemed(id);

            ticketNFT.transferFrom(msg.sender, address(this), id);
            depositor[id] = msg.sender;
            emit TicketEscrowed(id, msg.sender);
        }
    }

    function withdraw(uint256[] calldata ids) external {
        for (uint256 i = 0; i < ids.length; i++) {
            uint256 id = ids[i];
            if (depositor[id] != msg.sender) revert NotDepositor(id, msg.sender);

            depositor[id] = address(0);
            ticketNFT.transferFrom(address(this), msg.sender, id);
            emit TicketWithdrawn(id, msg.sender);
        }
    }

    function transferTicket(uint256 tokenId, address to) external {
        if (msg.sender != settlement) revert NotSettlement(msg.sender);
        depositor[tokenId] = address(0);
        ticketNFT.transferFrom(address(this), to, tokenId);
    }
}
```

```solidity
// src/IntentRegistry.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

contract IntentRegistry {
    struct Intent {
        address   owner;
        uint256[] offered;
        uint32    eventId;
        uint16    sessionMask;
        uint16    sectionMask;
        uint8     exactCount;
        bool      mustShareSession;
        bool      mustShareSection;
        bool      mustBeAdjacent;
        int256    maxNetPay;
        uint64    deadline;
        uint256   nonce;
    }

    // 0 = none, 1 = live, 2 = revoked, 3 = settled
    mapping(bytes32 => uint8) public state;
    mapping(bytes32 => address) public intentOwner;
    mapping(address => mapping(uint256 => bool)) public nonceUsed;

    bytes32 public immutable DOMAIN_SEPARATOR;

    bytes32 public constant INTENT_TYPEHASH = keccak256(
        "Intent(address owner,uint256[] offered,uint32 eventId,uint16 sessionMask,"
        "uint16 sectionMask,uint8 exactCount,bool mustShareSession,bool mustShareSection,"
        "bool mustBeAdjacent,int256 maxNetPay,uint64 deadline,uint256 nonce)"
    );

    error InvalidSignature(bytes32 intentHash);
    error IntentAlreadyExists(bytes32 intentHash);
    error NotIntentOwner(bytes32 intentHash, address caller);
    error IntentNotLive(bytes32 intentHash);
    error NonceAlreadyUsed(address owner, uint256 nonce);

    event IntentCommitted(bytes32 indexed intentHash, address indexed owner, uint256 nonce);
    event IntentRevoked(bytes32 indexed intentHash, address indexed owner);

    constructor() {
        DOMAIN_SEPARATOR = keccak256(abi.encode(
            keccak256("EIP712Domain(string name,string version,uint256 chainId,address verifyingContract)"),
            keccak256("RESHUFFLE"),
            keccak256("1"),
            block.chainid,
            address(this)
        ));
    }

    function hashIntent(Intent memory intent) public pure returns (bytes32) {
        return keccak256(abi.encode(
            INTENT_TYPEHASH,
            intent.owner,
            keccak256(abi.encodePacked(intent.offered)),
            intent.eventId,
            intent.sessionMask,
            intent.sectionMask,
            intent.exactCount,
            intent.mustShareSession,
            intent.mustShareSection,
            intent.mustBeAdjacent,
            intent.maxNetPay,
            intent.deadline,
            intent.nonce
        ));
    }

    function commit(Intent calldata intent, bytes calldata sig) external {
        if (nonceUsed[intent.owner][intent.nonce]) {
            revert NonceAlreadyUsed(intent.owner, intent.nonce);
        }

        bytes32 h = hashIntent(intent);
        if (state[h] != 0) revert IntentAlreadyExists(h);

        bytes32 digest = keccak256(abi.encodePacked("\x19\x01", DOMAIN_SEPARATOR, h));
        address recovered = _recover(digest, sig);
        if (recovered != intent.owner) revert InvalidSignature(h);

        state[h] = 1;
        intentOwner[h] = intent.owner;
        nonceUsed[intent.owner][intent.nonce] = true;

        emit IntentCommitted(h, intent.owner, intent.nonce);
    }

    function revoke(bytes32 intentHash) external {
        if (intentOwner[intentHash] != msg.sender) {
            revert NotIntentOwner(intentHash, msg.sender);
        }
        if (state[intentHash] != 1) revert IntentNotLive(intentHash);

        state[intentHash] = 2;
        emit IntentRevoked(intentHash, msg.sender);
    }

    function markSettled(bytes32 intentHash) external {
        // Settlement contract calls this — access control set during deployment
        state[intentHash] = 3;
    }

    function _recover(bytes32 digest, bytes calldata sig) internal pure returns (address) {
        if (sig.length != 65) return address(0);
        bytes32 r = bytes32(sig[0:32]);
        bytes32 s = bytes32(sig[32:64]);
        uint8 v = uint8(sig[64]);
        return ecrecover(digest, v, r, s);
    }
}
```

```solidity
            "\x19\x01",
            registry.DOMAIN_SEPARATOR(),
            h

        (uint8 v, bytes32 r, bytes32 s) = vm.sign(privateKey, digest);
        return abi.encodePacked(r, s, v);
```

```solidity
// src/Settlement.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "./TicketNFT.sol";
import "./Escrow.sol";
import "./IntentRegistry.sol";
import "@openzeppelin/contracts/token/ERC20/IERC20.sol";

contract Settlement {
    TicketNFT public immutable ticketNFT;
    Escrow public immutable escrow;
    IntentRegistry public immutable intentRegistry;
    IERC20 public immutable usdc;

    struct Leg {
        bytes32   intentHash;
        address   participant;
        uint256[] receives;
        int256    netPayment; // > 0 pays, < 0 receives
    }

    error IntentNotLive(bytes32 intentHash);
    error IntentExpired(bytes32 intentHash, uint64 deadline);
    error SignatureMismatch(bytes32 intentHash);
    error TicketNotEscrowed(uint256 tokenId, address expectedOwner);
    error TicketRedeemed(uint256 tokenId);
    error ConservationViolated();
    error EventMismatch(bytes32 intentHash, uint256 tokenId);
    error SessionNotAccepted(bytes32 intentHash, uint16 sessionId);
    error SectionNotAccepted(bytes32 intentHash, uint16 sectionId);
    error CountMismatch(bytes32 intentHash, uint8 expected, uint8 actual);
    error NotSameSession(bytes32 intentHash);
    error NotSameSection(bytes32 intentHash);
    error SeatsNotAdjacent(bytes32 intentHash);
    error BudgetExceeded(bytes32 intentHash, int256 limit, int256 actual);
    error PaymentImbalance(int256 total);
    error InsufficientPaymentCapacity(address payer, uint256 required);

    event Settled(
        bytes32[] intentHashes,
        address[] participants,
        uint256 ticketCount,
        uint256 timestamp
    );

    constructor(address _nft, address _escrow, address _registry, address _usdc) {
        ticketNFT = TicketNFT(_nft);
        escrow = Escrow(_escrow);
        intentRegistry = IntentRegistry(_registry);
        usdc = IERC20(_usdc);
    }

    function settle(
        IntentRegistry.Intent[] calldata intents,
        Leg[] calldata legs
    ) external {
        // --- V1: intent validity ---
        for (uint256 i = 0; i < intents.length; i++) {
            bytes32 h = intentRegistry.hashIntent(intents[i]);
            if (intentRegistry.state(h) != 1) revert IntentNotLive(h);
            if (block.timestamp > intents[i].deadline) {
                revert IntentExpired(h, intents[i].deadline);
            }
            if (intentRegistry.intentOwner(h) != intents[i].owner) {
                revert SignatureMismatch(h);
            }
        }

        // --- V2: escrow ownership ---
        uint256 totalOffered = 0;
        for (uint256 i = 0; i < intents.length; i++) {
            for (uint256 j = 0; j < intents[i].offered.length; j++) {
                uint256 tokenId = intents[i].offered[j];
                if (escrow.depositor(tokenId) != intents[i].owner) {
                    revert TicketNotEscrowed(tokenId, intents[i].owner);
                }
                totalOffered++;
            }
        }

        // --- V3: ticket status ---
        for (uint256 i = 0; i < intents.length; i++) {
            for (uint256 j = 0; j < intents[i].offered.length; j++) {
                (,,,,, uint8 status) = ticketNFT.meta(intents[i].offered[j]);
                if (status == 1) revert TicketRedeemed(intents[i].offered[j]);
            }
        }

        // --- V4: conservation ---
        uint256 totalReceived = 0;
        for (uint256 i = 0; i < legs.length; i++) {
            totalReceived += legs[i].receives.length;
        }
        if (totalOffered != totalReceived) revert ConservationViolated();

        // Build offered set and verify bijection
        // Every offered ticket must appear in exactly one leg's receives
        // Every received ticket must appear in exactly one intent's offered
        {
            // Mark all offered tickets
            mapping(uint256 => bool) storage _used; // cannot use storage mapping in memory
            // Use a different approach: iterate and verify
            // For each received ticket, verify it was offered by some intent
            // For each offered ticket, verify it appears in some leg's receives
            // At demo scale this is O(n^2), acceptable

            // Check: every received ticket is in some intent's offered
            for (uint256 i = 0; i < legs.length; i++) {
                for (uint256 j = 0; j < legs[i].receives.length; j++) {
                    uint256 tokenId = legs[i].receives[j];
                    bool found = false;
                    for (uint256 k = 0; k < intents.length; k++) {
                        for (uint256 l = 0; l < intents[k].offered.length; l++) {
                            if (intents[k].offered[l] == tokenId) {
                                found = true;
                                break;
                            }
                        }
                        if (found) break;
                    }
                    if (!found) revert ConservationViolated();
                }
            }

            // Check: every offered ticket appears in some leg's receives
            for (uint256 i = 0; i < intents.length; i++) {
                for (uint256 j = 0; j < intents[i].offered.length; j++) {
                    uint256 tokenId = intents[i].offered[j];
                    bool found = false;
                    for (uint256 k = 0; k < legs.length; k++) {
                        for (uint256 l = 0; l < legs[k].receives.length; l++) {
                            if (legs[k].receives[l] == tokenId) {
                                found = true;
                                break;
                            }
                        }
                        if (found) break;
                    }
                    if (!found) revert ConservationViolated();
                }
            }
        }

        // --- V5: per-participant constraints ---
        for (uint256 i = 0; i < legs.length; i++) {
            Leg calldata leg = legs[i];
            // Find the matching intent
            IntentRegistry.Intent calldata intent = intents[i];
            bytes32 h = intentRegistry.hashIntent(intent);

            // exactCount
            if (leg.receives.length != intent.exactCount) {
                revert CountMismatch(h, intent.exactCount, uint8(leg.receives.length));
            }

            uint16 firstSession = 0;
            uint16 firstSection = 0;

            for (uint256 j = 0; j < leg.receives.length; j++) {
                uint256 tokenId = leg.receives[j];
                (uint32 eventId, uint16 sessionId, uint16 sectionId, uint16 row, uint16 seat,) =
                    ticketNFT.meta(tokenId);

                // eventId
                if (eventId != intent.eventId) revert EventMismatch(h, tokenId);

                // sessionMask
                if (intent.sessionMask & (uint16(1) << sessionId) == 0) {
                    revert SessionNotAccepted(h, sessionId);
                }

                // sectionMask
                if (intent.sectionMask & (uint16(1) << sectionId) == 0) {
                    revert SectionNotAccepted(h, sectionId);
                }

                // cohesion: mustShareSession
                if (intent.mustShareSession) {
                    if (j == 0) firstSession = sessionId;
                    else if (sessionId != firstSession) revert NotSameSession(h);
                }

                // cohesion: mustShareSection
                if (intent.mustShareSection) {
                    if (j == 0) firstSection = sectionId;
                    else if (sectionId != firstSection) revert NotSameSection(h);
                }
            }

            // adjacency check
            if (intent.mustBeAdjacent && leg.receives.length > 1) {
                _checkAdjacency(h, leg.receives);
            }

            // --- V6: budget ---
            if (intent.maxNetPay >= 0) {
                if (leg.netPayment > intent.maxNetPay) {
                    revert BudgetExceeded(h, intent.maxNetPay, leg.netPayment);
                }
            } else {
                if (leg.netPayment < intent.maxNetPay) {
                    revert BudgetExceeded(h, intent.maxNetPay, leg.netPayment);
                }
            }
        }

        // --- V7: payment balance ---
        int256 total = 0;
        for (uint256 i = 0; i < legs.length; i++) {
            total += legs[i].netPayment;
        }
        if (total != 0) revert PaymentImbalance(total);

        // --- V8: payment capacity ---
        for (uint256 i = 0; i < legs.length; i++) {
            if (legs[i].netPayment > 0) {
                uint256 required = uint256(legs[i].netPayment);
                address payer = legs[i].participant;
                if (usdc.balanceOf(payer) < required) {
                    revert InsufficientPaymentCapacity(payer, required);
                }
                if (usdc.allowance(payer, address(this)) < required) {
                    revert InsufficientPaymentCapacity(payer, required);
                }
            }
        }

        // --- Effects: transfer tickets ---
        for (uint256 i = 0; i < legs.length; i++) {
            for (uint256 j = 0; j < legs[i].receives.length; j++) {
                escrow.transferTicket(legs[i].receives[j], legs[i].participant);
            }
        }

        // --- Effects: transfer USDC ---
        for (uint256 i = 0; i < legs.length; i++) {
            if (legs[i].netPayment > 0) {
                usdc.transferFrom(legs[i].participant, address(this), uint256(legs[i].netPayment));
            }
        }
        for (uint256 i = 0; i < legs.length; i++) {
            if (legs[i].netPayment < 0) {
                usdc.transfer(legs[i].participant, uint256(-legs[i].netPayment));
            }
        }

        // --- Effects: mark intents settled ---
        bytes32[] memory hashes = new bytes32[](intents.length);
        address[] memory participants = new address[](legs.length);
        for (uint256 i = 0; i < intents.length; i++) {
            bytes32 h = intentRegistry.hashIntent(intents[i]);
            intentRegistry.markSettled(h);
            hashes[i] = h;
        }
        for (uint256 i = 0; i < legs.length; i++) {
            participants[i] = legs[i].participant;
        }

        emit Settled(hashes, participants, totalOffered, block.timestamp);
    }

    function _checkAdjacency(bytes32 intentHash, uint256[] calldata tokenIds) internal view {
        // All same section, same row, consecutive seats
        (, uint16 firstSession, uint16 firstSection, uint16 firstRow, uint16 firstSeat,) =
            ticketNFT.meta(tokenIds[0]);

        uint16[] memory seats = new uint16[](tokenIds.length);
        seats[0] = firstSeat;

        for (uint256 i = 1; i < tokenIds.length; i++) {
            (, uint16 sid, uint16 secId, uint16 row, uint16 seat,) = ticketNFT.meta(tokenIds[i]);
            if (secId != firstSection || row != firstRow) {
                revert SeatsNotAdjacent(intentHash);
            }
            seats[i] = seat;
        }

        // Sort seats (insertion sort, small arrays)
        for (uint256 i = 1; i < seats.length; i++) {
            uint16 key = seats[i];
            uint256 j = i;
            while (j > 0 && seats[j - 1] > key) {
                seats[j] = seats[j - 1];
                j--;
            }
            seats[j] = key;
        }

        // Check consecutive
        for (uint256 i = 1; i < seats.length; i++) {
            if (seats[i] != seats[i - 1] + 1) {
                revert SeatsNotAdjacent(intentHash);
            }
        }
    }
}
```

```typescript
// solver/src/constraints.ts
import { Intent, Leg, TicketMeta } from './types.js';

export type ConstraintError =
  | { type: 'count_mismatch'; expected: number; actual: number }
  | { type: 'session_not_accepted'; sessionId: number }
  | { type: 'section_not_accepted'; sectionId: number }
  | { type: 'event_mismatch'; tokenId: bigint }
  | { type: 'not_same_session' }
  | { type: 'not_same_section' }
  | { type: 'seats_not_adjacent' }
  | { type: 'budget_exceeded'; limit: bigint; actual: bigint };

export function validateLeg(
  intent: Intent,
  leg: Leg,
  ticketMeta: Map<bigint, TicketMeta>,
): ConstraintError | null {
  // exactCount
  if (leg.receives.length !== intent.exactCount) {
    return { type: 'count_mismatch', expected: intent.exactCount, actual: leg.receives.length };
  }

  let firstSession = -1;
  let firstSection = -1;

  for (const tokenId of leg.receives) {
    const meta = ticketMeta.get(tokenId);
    if (!meta) throw new Error(`Unknown ticket ${tokenId}`);

    if (meta.eventId !== intent.eventId) {
      return { type: 'event_mismatch', tokenId };
    }

    if ((intent.sessionMask & (1 << meta.sessionId)) === 0) {
      return { type: 'session_not_accepted', sessionId: meta.sessionId };
    }

    if ((intent.sectionMask & (1 << meta.sectionId)) === 0) {
      return { type: 'section_not_accepted', sectionId: meta.sectionId };
    }

    if (intent.mustShareSession) {
      if (firstSession === -1) firstSession = meta.sessionId;
      else if (meta.sessionId !== firstSession) return { type: 'not_same_session' };
    }

    if (intent.mustShareSection) {
      if (firstSection === -1) firstSection = meta.sectionId;
      else if (meta.sectionId !== firstSection) return { type: 'not_same_section' };
    }
  }

  // adjacency
  if (intent.mustBeAdjacent && leg.receives.length > 1) {
    const metas = leg.receives.map(id => ticketMeta.get(id)!);
    const section = metas[0].sectionId;
    const row = metas[0].row;

    if (!metas.every(m => m.sectionId === section && m.row === row)) {
      return { type: 'seats_not_adjacent' };
    }

    const seats = metas.map(m => m.seat).sort((a, b) => a - b);
    for (let i = 1; i < seats.length; i++) {
      if (seats[i] !== seats[i - 1] + 1) return { type: 'seats_not_adjacent' };
    }
  }

  // budget
  if (intent.maxNetPay >= 0n) {
    if (leg.netPayment > intent.maxNetPay) {
      return { type: 'budget_exceeded', limit: intent.maxNetPay, actual: leg.netPayment };
    }
  } else {
    if (leg.netPayment < intent.maxNetPay) {
      return { type: 'budget_exceeded', limit: intent.maxNetPay, actual: leg.netPayment };
    }
  }

  return null;
}
```

```typescript
// solver/src/search.ts
import { Intent, Leg, Reshuffle, TicketMeta } from './types.js';
import { validateLeg } from './constraints.js';

export interface SearchConfig {
  participantCap: number;
  candidateCap: number;
  timeoutMs: number;
}

export const DEFAULT_CONFIG: SearchConfig = {
  participantCap: 20,
  candidateCap: 1000,
  timeoutMs: 10_000,
};

// Bounded exhaustive search: enumerate subsets of intents, for each subset
// try all valid ticket assignments.
export function search(
  intents: Intent[],
  ticketMeta: Map<bigint, TicketMeta>,
  config: SearchConfig = DEFAULT_CONFIG,
): Reshuffle[] {
  const results: Reshuffle[] = [];
  const start = Date.now();

  const capped = intents.slice(0, config.participantCap);

  // Enumerate subsets of size >= 2
  for (let size = 2; size <= capped.length && results.length < config.candidateCap; size++) {
    if (Date.now() - start > config.timeoutMs) break;

    for (const subset of combinations(capped, size)) {
      if (Date.now() - start > config.timeoutMs) break;
      if (results.length >= config.candidateCap) break;

      // Collect all offered tickets from this subset
      const allOffered = subset.flatMap(i => i.offered);

      // Try to assign tickets to legs
      const reshuffle = tryAssign(subset, allOffered, ticketMeta);
      if (reshuffle) results.push(reshuffle);
    }
  }

  return results;
}

function tryAssign(
  intents: Intent[],
  allTickets: bigint[],
  ticketMeta: Map<bigint, TicketMeta>,
): Reshuffle | null {
  // For each intent, find valid bundles from allTickets (excluding own offered)
  // This is a constraint satisfaction problem; at demo scale, backtracking works

  const legs: Leg[] = [];
  const used = new Set<bigint>();

  function backtrack(idx: number): boolean {
    if (idx === intents.length) {
      // Check conservation: all offered tickets must be used
      return used.size === allTickets.length;
    }

    const intent = intents[idx];
    const ownOffered = new Set(intent.offered.map(String));
    const available = allTickets.filter(
      t => !used.has(t) && !ownOffered.has(String(t))
    );

    if (intent.exactCount === 0) {
      // Pure seller — receives nothing
      const leg: Leg = {
        intentHash: '', // computed later
        participant: intent.owner,
        receives: [],
        netPayment: 0n, // computed during payment assignment
      };
      legs.push(leg);
      const result = backtrack(idx + 1);
      if (result) return true;
      legs.pop();
      return false;
    }

    // Find all valid bundles of size exactCount
    for (const bundle of combinations(available, intent.exactCount)) {
      const leg: Leg = {
        intentHash: '',
        participant: intent.owner,
        receives: bundle.map(t => t),
        netPayment: 0n,
      };

      const err = validateLeg(intent, leg, ticketMeta);
      if (err) continue;

      for (const t of bundle) used.add(t);
      legs.push(leg);

      if (backtrack(idx + 1)) return true;

      legs.pop();
      for (const t of bundle) used.delete(t);
    }

    return false;
  }

  if (!backtrack(0)) return null;

  // Assign payments: minimize total net payment, respecting budgets
  // Simple: everyone pays 0 if budgets allow
  const totalNet = legs.reduce((s, l) => s + l.netPayment, 0n);
  
  return {
    intents: [...intents],
    legs: [...legs],
    totalNetPayment: totalNet,
    participantCount: intents.length,
  };
}

function* combinations<T>(arr: T[], k: number): Generator<T[]> {
  if (k === 0) { yield []; return; }
  if (k > arr.length) return;

  for (let i = 0; i <= arr.length - k; i++) {
    for (const rest of combinations(arr.slice(i + 1), k - 1)) {
      yield [arr[i], ...rest];
    }
  }
}
```

```typescript
// solver/src/rank.ts
import { Reshuffle } from './types.js';

// Published ranking rule:
// Among valid reshuffles, minimise total absolute net payment.
// Ties break toward fewer participants, then lower estimated gas.
export function rank(candidates: Reshuffle[]): Reshuffle[] {
  return [...candidates].sort((a, b) => {
    // 1. Minimise total absolute net payment
    const aTotal = totalAbsPayment(a);
    const bTotal = totalAbsPayment(b);
    if (aTotal !== bTotal) return aTotal < bTotal ? -1 : 1;

    // 2. Fewer participants
    if (a.participantCount !== b.participantCount) {
      return a.participantCount - b.participantCount;
    }

    // 3. Fewer tickets (proxy for gas)
    const aTickets = a.legs.reduce((s, l) => s + l.receives.length, 0);
    const bTickets = b.legs.reduce((s, l) => s + l.receives.length, 0);
    return aTickets - bTickets;
  });
}

function totalAbsPayment(r: Reshuffle): bigint {
  let sum = 0n;
  for (const leg of r.legs) {
    sum += leg.netPayment < 0n ? -leg.netPayment : leg.netPayment;
  }
  return sum;
}
```
